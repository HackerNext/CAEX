// GEMINI API HELPER WITH EXPONENTIAL BACKOFF
// ==========================================
async function callGeminiAPI({ systemPrompt, userPrompt, inlineFiles = [], jsonSchema = null, customApiKey = '' }) {
  const apiKey = customApiKey ? customApiKey.trim() : "";
  const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;

  const parts = [{ text: userPrompt }];
  for (const file of inlineFiles) {
    if (file.base64 && file.mimeType) {
      parts.push({
        inlineData: {
          mimeType: file.mimeType,
          data: file.base64
        }
      });
    }
  }

  const payload = {
    contents: [{ role: 'user', parts }],
    systemInstruction: {
      parts: [{ text: systemPrompt }]
    }
  };

  if (jsonSchema) {
    payload.generationConfig = {
      responseMimeType: 'application/json',
      responseSchema: jsonSchema
    };
  }

  const delays = [1000, 2000, 4000];
  let lastError = null;

  for (let attempt = 0; attempt <= delays.length; attempt++) {
    try {
      const response = await fetch(apiUrl, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      });

      if (!response.ok) {
        const errText = await response.text();
        throw new Error(`API status ${response.status}: ${errText}`);
      }

      const result = await response.json();
      const text = result?.candidates?.[0]?.content?.parts?.[0]?.text;
      if (!text) throw new Error('Empty response from Gemini evaluator.');

      return jsonSchema ? JSON.parse(text) : text;
    } catch (err) {
      lastError = err;
      if (attempt < delays.length) {
        await new Promise((r) => setTimeout(r, delays[attempt]));
      }
    }
  }

  throw lastError;
}

// File to Base64 utility
function fileToBase64(file) {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.onload = () => {
      const resultStr = reader.result;
      const base64 = resultStr.split(',')[1];
      resolve({
        name: file.name,
        mimeType: file.type || 'application/pdf',
        base64,
        previewUrl: file.type?.startsWith('image/') ? resultStr : null,
        sizeKB: Math.round(file.size / 1024)
      });
    };
    reader.onerror = reject;
    reader.readAsDataURL(file);
  });
}

        accentBg: 'bg-[#D97757]',
        accentHover: 'hover:bg-[#C46647]',

  if (activeQuiz) {

    const currentQ = questions[currentQIndex] || questions[0];

    return (

        <style>{`

          .font-serif-claude { font-family: 'Newsreader', 'Instrument Serif', Georgia, serif; }

        `}</style>

        <header className={`sticky top-0 z-30 border-b ${theme.border} ${theme.navBg} backdrop-blur-md px-4 py-3`}>

                className={`p-2 rounded-xl border ${theme.border} ${theme.card} hover:opacity-80 transition`}
                title="Exit Quiz"
              >
                <ChevronLeft className="w-4 h-4" />
              </button>
              <div className="truncate">
                <div className="flex items-center gap-2">
                  <span className={`text-[11px] font-mono-claude uppercase tracking-wider px-2 py-0.5 rounded-md border ${theme.accentSoft}`}>
                    {activeQuiz.level}
                  </span>
                  <span className={`text-xs ${theme.textSecondary} truncate`}>
                    {activeQuiz.negativeMarking > 0 ? `+1 / -${activeQuiz.negativeMarking} Marks` : '+1 Mark (No Neg)'}
                  </span>
                </div>
                <h1 className="font-serif-claude text-base sm:text-lg font-medium truncate mt-0.5">
                  {activeQuiz.title}
                </h1>
              </div>
            </div>

            <div className="flex items-center gap-2 shrink-0">
              {!quizResult && (
                <div className={`flex items-center gap-1.5 px-2.5 py-1.5 rounded-xl border ${theme.border} ${theme.card} font-mono-claude text-xs`}>
                  <Clock className="w-3.5 h-3.5 text-[#D97757]" />

                </div>
              )}
              {!quizResult && (
                <button
                  onClick={() => setShowReviewDrawer(true)}
                  className={`px-3 py-1.5 rounded-xl text-xs font-medium border ${theme.border} ${theme.cardElevated} flex items-center gap-1.5 shadow-sm`}
                >
                  <ListChecks className="w-4 h-4 text-[#D97757]" />
                  <span>Review ({attemptedCount}/{questions.length})</span>
                </button>
              )}
            </div>
          </div>
        </header>

        {/* Main Quiz Body OR Post-Submission Scorecard */}
        <main className="flex-1 max-w-3xl w-full mx-auto px-4 py-5 pb-28">
          {!quizResult ? (
            <div className="space-y-5">
              {/* Question Progress Strip */}
              <div className="flex items-center justify-between text-xs">
                <span className={`font-mono-claude ${theme.textSecondary}`}>
                  QUESTION {currentQIndex + 1} OF {questions.length}
                </span>
                {currentQ.chapter && (
                  <span className={`px-2.5 py-0.5 rounded-full border ${theme.border} ${theme.card} ${theme.textSecondary} truncate max-w-[220px] sm:max-w-xs`}>
                    {currentQ.chapter}
                  </span>
                )}
              </div>

              {/* Question Number Quick Pills */}
              <div className="flex items-center gap-1.5 overflow-x-auto pb-1.5 no-scrollbar">
                {questions.map((q, idx) => {
                  const isAnswered = userResponses[idx] !== undefined;
                  const isReview = markedForReview[idx];
                  const isCurrent = idx === currentQIndex;
                  return (
                    <button
                      key={idx}
                      onClick={() => setCurrentQIndex(idx)}
                      className={`w-8 h-8 rounded-lg font-mono-claude text-xs font-medium shrink-0 flex items-center justify-center border transition-all ${
                        isCurrent
                          ? 'ring-2 ring-[#D97757] border-[#D97757] font-bold'
                          : theme.border
                      } ${
                        isReview
                          ? 'bg-amber-500/20 text-amber-600 dark:text-amber-400 border-amber-500/40'
                          : isAnswered
                          ? 'bg-emerald-600/20 text-emerald-700 dark:text-emerald-300 border-emerald-500/40'
                          : theme.card
                      }`}
                    >
                      {idx + 1}
                    </button>
                  );
                })}
              </div>

              {/* Question Statement Card */}
              <div className={`p-5 sm:p-6 rounded-2xl border ${theme.border} ${theme.cardElevated} shadow-sm space-y-5`}>
                <div className="flex items-start justify-between gap-3">
                  <span className="font-mono-claude text-xs font-semibold px-2.5 py-1 rounded-lg bg-[#D97757]/15 text-[#D97757]">
                    {currentQ.qNo || `Q${currentQIndex + 1}`}
                  </span>
                  <button
                    onClick={() => handleToggleMarkReview(currentQIndex)}
                    className={`flex items-center gap-1.5 text-xs px-3 py-1 rounded-lg border transition ${
                      markedForReview[currentQIndex]
                        ? 'bg-amber-500/20 border-amber-500/50 text-amber-600 dark:text-amber-400 font-medium'
                        : `${theme.border} ${theme.textSecondary}`
                    }`}
                  >
                    <Flag className="w-3.5 h-3.5" />
                    <span>{markedForReview[currentQIndex] ? 'Marked for Review' : 'Mark for Review'}</span>
                  </button>
                </div>

                <p className="font-serif-claude text-lg sm:text-xl leading-relaxed whitespace-pre-line">
                  {currentQ.question}
                </p>

                {/* Options List */}
                <div className="space-y-2.5 pt-2">
                  {(currentQ.options || []).map((optText, optIdx) => {
                    const isSelected = userResponses[currentQIndex] === optIdx;
                    const optionLabel = ['A', 'B', 'C', 'D'][optIdx] || optIdx + 1;

                    return (
                      <button
                        key={optIdx}
                        onClick={() => handleSelectOption(currentQIndex, optIdx)}
                        className={`w-full text-left p-3.5 sm:p-4 rounded-xl border transition-all flex items-start gap-3.5 ${
                          isSelected
                            ? 'border-[#D97757] bg-[#D97757]/12 shadow-sm'
                            : `${theme.border} ${theme.card} hover:opacity-90`
                        }`}
                      >
                        <span
                          className={`w-7 h-7 rounded-lg font-mono-claude text-xs font-semibold flex items-center justify-center shrink-0 mt-0.5 ${
                            isSelected
                              ? 'bg-[#D97757] text-white'
                              : `border ${theme.border} ${theme.cardElevated}`
                          }`}
                        >
                          {optionLabel}
                        </span>
                        <span className="text-sm sm:text-base leading-relaxed flex-1">
                          {optText}
                        </span>
                      </button>
                    );
                  })}
                </div>

                {/* Clear Response Row */}
                <div className="flex items-center justify-between pt-2 border-t border-dashed border-[#E5E2DC] dark:border-[#2E2D2A]">
                  <button
                    onClick={() => handleClearOption(currentQIndex)}
                    disabled={userResponses[currentQIndex] === undefined}
                    className={`text-xs flex items-center gap-1.5 ${
                      userResponses[currentQIndex] !== undefined
                        ? 'text-[#D97757] hover:underline'
                        : `${theme.textSecondary} opacity-40 cursor-not-allowed`
                    }`}
                  >
                    <RotateCcw className="w-3.5 h-3.5" />
                    <span>Clear Selected Response</span>
                  </button>
                  <span className={`text-xs font-mono-claude ${theme.textSecondary}`}>
                    {userResponses[currentQIndex] !== undefined
                      ? `Selected: Option ${['A', 'B', 'C', 'D'][userResponses[currentQIndex]]}`
                      : 'Not Attempted'}
                  </span>
                </div>
              </div>

              {/* Bottom Navigation Controls for Quiz */}
              <div className="flex items-center justify-between gap-3 pt-2">
                <button
                  onClick={() => setCurrentQIndex((i) => Math.max(0, i - 1))}
                  disabled={currentQIndex === 0}
                  className={`px-4 py-3 rounded-xl border ${theme.border} ${theme.card} text-sm font-medium flex items-center gap-1.5 disabled:opacity-40`}
                >
                  <ChevronLeft className="w-4 h-4" />
                  <span>Previous</span>
                </button>

                <button
                  onClick={() => setShowReviewDrawer(true)}
                  className={`px-4 py-3 rounded-xl border border-[#D97757]/40 bg-[#D97757]/10 text-[#D97757] text-sm font-semibold flex items-center gap-2`}
                >
                  <Eye className="w-4 h-4" />
                  <span>Check Answers & Submit</span>
                </button>

                {currentQIndex < questions.length - 1 ? (
                  <button
                    onClick={() => setCurrentQIndex((i) => Math.min(questions.length - 1, i + 1))}
                    className={`px-5 py-3 rounded-xl ${theme.accentBg} text-white text-sm font-medium flex items-center gap-1.5 shadow-sm`}
                  >
                    <span>Next</span>
                    <ChevronRight className="w-4 h-4" />
                  </button>
                ) : (
                  <button
                    onClick={() => setShowReviewDrawer(true)}
                    className={`px-5 py-3 rounded-xl ${theme.accentBg} text-white text-sm font-semibold flex items-center gap-1.5 shadow-sm`}
                  >
                    <span>Finish Review</span>
                    <CheckCircle2 className="w-4 h-4" />
                  </button>
                )}
              </div>
            </div>
          ) : (
            /* ==========================================
               POST-SUBMISSION ICAI QUIZ SCORECARD & EXPLANATIONS
               ========================================== */
            <div className="space-y-6 animate-fadeIn">
              {/* Top Result Summary Banner */}
              <div className={`p-6 rounded-2xl border ${theme.border} ${theme.cardElevated} shadow-sm space-y-5`}>
                <div className="flex flex-wrap items-center justify-between gap-4">
                  <div>
                    <span className={`text-xs font-mono-claude uppercase tracking-wider px-2.5 py-1 rounded-md border ${theme.accentSoft}`}>
                      ICAI OMR Evaluation Complete
                    </span>
                    <h2 className="font-serif-claude text-2xl sm:text-3xl font-medium mt-2">
                      {quizResult.title}
                    </h2>
                    <p className={`text-xs ${theme.textSecondary} mt-1`}>
                      Time Taken: {formatTime(quizResult.timeTakenSecs)} â¢ Negative Marking: -{quizResult.negativeMarking} per wrong answer
                    </p>
                  </div>

                  <div className={`px-5 py-4 rounded-2xl border ${theme.border} ${theme.card} text-center min-w-[140px]`}>
                    <div className="text-xs font-mono-claude uppercase text-[#D97757]">Net ICAI Score</div>
                    <div className="font-mono-claude text-3xl font-bold mt-1">
                      {quizResult.netMarks} <span className={`text-base font-normal ${theme.textSecondary}`}>/ {quizResult.maxMarks}</span>
                    </div>
                    <div className={`text-xs font-medium mt-1 ${quizResult.percentage >= 50 ? 'text-emerald-600 dark:text-emerald-400' : 'text-amber-600'}`}>
                      {quizResult.percentage}% ({quizResult.percentage >= 50 ? 'PASS - ICAI 50% Criteria' : 'Needs Revision'})
                    </div>
                  </div>
                </div>

                {/* 4 Metric Pills */}
                <div className="grid grid-cols-2 sm:grid-cols-4 gap-2.5 pt-2">
                  <div className={`p-3 rounded-xl border ${theme.border} ${theme.card}`}>
                    <div className={`text-xs ${theme.textSecondary}`}>Correct (+1.00)</div>
                    <div className="font-mono-claude text-lg font-semibold text-emerald-600 dark:text-emerald-400 mt-0.5">
                      {quizResult.correctCount} (+{quizResult.rawMarks})
                    </div>
                  </div>
                  <div className={`p-3 rounded-xl border ${theme.border} ${theme.card}`}>
                    <div className={`text-xs ${theme.textSecondary}`}>Incorrect (-{quizResult.negativeMarking})</div>
                    <div className="font-mono-claude text-lg font-semibold text-rose-600 dark:text-rose-400 mt-0.5">
                      {quizResult.wrongCount} (-{quizResult.negativeDeducted})
                    </div>
                  </div>
                  <div className={`p-3 rounded-xl border ${theme.border} ${theme.card}`}>
                    <div className={`text-xs ${theme.textSecondary}`}>Skipped (0.00)</div>

                        ? Math.round((quizResult.correctCount / (quizResult.correctCount + quizResult.wrongCount)) * 100)
                        : 0}%
                    </div>
                  </div>

                <div className="flex flex-wrap gap-2.5 pt-1">
                  <button
                    onClick={() => startQuizSession(activeQuiz)}
                    className={`px-4 py-2.5 rounded-xl border ${theme.border} ${theme.card} text-xs font-semibold flex items-center gap-1.5`}

                    <RotateCcw className="w-3.5 h-3.5" />
                    <span>Re-attempt Paper</span>
                  </button>
                  <button
                    onClick={() => {
                      setActiveQuiz(null);
                      setQuizResult(null);

                    className={`px-4 py-2.5 rounded-xl ${theme.accentBg} text-white text-xs font-semibold flex items-center gap-1.5`}

                    <span>Back to Dashboard</span>
                  </button>
                </div>
              </div>

              {/* Detailed Question-by-Question Explanation List */}
              <div className="space-y-4">
                <h3 className="font-serif-claude text-xl font-medium px-1">
                  Question-by-Question ICAI Module Solutions

                {quizResult.questions.map((q, idx) => {
                  const userChoice = quizResult.userResponses[idx];
                  const isSkipped = userChoice === undefined;
                  const isCorrect = userChoice === q.correctIndex;
                  const optionLetters = ['A', 'B', 'C', 'D'];

                  return (
                    <div
                      key={idx}
                      className={`p-5 rounded-2xl border ${theme.border} ${theme.cardElevated} space-y-3.5`}
                    >
                      <div className="flex items-center justify-between gap-2">
                        <div className="flex items-center gap-2">
                          <span className="font-mono-claude text-xs font-bold px-2.5 py-1 rounded-md bg-[#D97757]/15 text-[#D97757]">

                          </span>

                            <span className={`text-xs ${theme.textSecondary}`}>{q.chapter}</span>
                          )}
                        </div>
                        <span
                          className={`text-xs font-mono-claude font-semibold px-2.5 py-1 rounded-lg ${
                            isSkipped
                              ? 'bg-stone-500/15 text-stone-500'
                              : isCorrect
                              ? 'bg-emerald-500/15 text-emerald-600 dark:text-emerald-400'
                              : 'bg-rose-500/15 text-rose-600 dark:text-rose-400'
                          }`}
                        >
                          {isSkipped
                            ? 'Skipped (0M)'
                            : isCorrect
                            ? '+1.00 Mark'
                            : `-${quizResult.negativeMarking} Mark`}
                        </span>
                      </div>

                      <p className="font-serif-claude text-base sm:text-lg leading-relaxed">
                        {q.question}

                            <div

                              className={`p-3 rounded-xl border text-xs sm:text-sm flex items-start gap-2.5 ${
                                isRightOpt
                                  ? 'border-emerald-500/60 bg-emerald-500/10 font-medium'
                                  : isUserOpt && !isRightOpt
                                  ? 'border-rose-500/60 bg-rose-500/10'
                                  : `${theme.border} ${theme.card}`
                              }`}
                            >
                              <span className="font-mono-claude font-bold shrink-0">
                                ({optionLetters[oIdx]})
                              </span>
                              <span className="flex-1">{opt}</span>
                              {isRightOpt && <Check className="w-4 h-4 text-emerald-600 shrink-0" />}
                              {isUserOpt && !isRightOpt && <X className="w-4 h-4 text-rose-600 shrink-0" />}
                            </div>
                          );
                        })}
                      </div>

                      {/* ICAI Explanation Box */}
                      <div className={`p-3.5 rounded-xl border ${theme.border} ${theme.card} text-xs sm:text-sm space-y-1`}>
                        <div className="font-semibold text-[#D97757] flex items-center gap-1.5">
                          <BookOpen className="w-3.5 h-3.5" />
                          <span>ICAI Study Material Explanation (Correct Answer: Option {optionLetters[q.correctIndex]})</span>
                        </div>
                        <p className={`${theme.textSecondary} leading-relaxed`}>
                          {q.explanation}
                        </p>
                      </div>
                    </div>
                  );
                })}
              </div>
            </div>
          )}
        </main>

        {/* ==========================================
            PRE-SUBMISSION REVIEW DRAWER / MODAL
           ========================================== */}
        {showReviewDrawer && !quizResult && (
          <div className="fixed inset-0 z-50 bg-black/60 backdrop-blur-xs flex items-end sm:items-center justify-center p-0 sm:p-4">
            <div className={`w-full max-w-2xl max-h-[88vh] flex flex-col rounded-t-3xl sm:rounded-3xl border ${theme.border} ${theme.cardElevated} shadow-2xl overflow-hidden`}>
              {/* Modal Header */}
              <div className={`px-5 py-4 border-b ${theme.border} flex items-center justify-between`}>
                <div>
                  <h3 className="font-serif-claude text-xl font-medium">
                    Pre-Submission Answer Sheet Review
                  </h3>
                  <p className={`text-xs ${theme.textSecondary}`}>
                    Verify all your attempted and skipped questions before final OMR submission
                  </p>
                </div>
                <button
                  onClick={() => setShowReviewDrawer(false)}
                  className={`p-2 rounded-xl border ${theme.border} ${theme.card}`}
                >
                  <X className="w-4 h-4" />
                </button>
              </div>

              {/* Status Summary Strip */}
              <div className={`px-5 py-3 border-b ${theme.border} ${theme.card} grid grid-cols-3 gap-2 text-center`}>
                <div className="p-2 rounded-xl bg-emerald-500/10 border border-emerald-500/30">
                  <div className="font-mono-claude text-base font-bold text-emerald-600 dark:text-emerald-400">
                    {attemptedCount}
                  </div>
                  <div className="text-[11px] text-emerald-700 dark:text-emerald-300">Attempted</div>
                </div>
                <div className="p-2 rounded-xl bg-stone-500/10 border border-stone-500/30">
                  <div className="font-mono-claude text-base font-bold">
                    {skippedCount}
                  </div>
                  <div className={`text-[11px] ${theme.textSecondary}`}>Unattempted / Skipped</div>
                </div>
                <div className="p-2 rounded-xl bg-amber-500/10 border border-amber-500/30">
                  <div className="font-mono-claude text-base font-bold text-amber-600 dark:text-amber-400">
                    {reviewCount}
                  </div>
                  <div className="text-[11px] text-amber-700 dark:text-amber-300">Marked for Review</div>
                </div>
              </div>

              {/* Scrollable Question & Checked Answer List */}
              <div className="flex-1 overflow-y-auto p-5 space-y-2.5">
                {questions.map((q, idx) => {
                  const chosen = userResponses[idx];
                  const isFlagged = markedForReview[idx];
                  const optLetter = chosen !== undefined ? ['A', 'B', 'C', 'D'][chosen] : null;

                  return (
                    <div
                      key={idx}
                      onClick={() => {
                        setCurrentQIndex(idx);
                        setShowReviewDrawer(false);
                      }}
                      className={`p-3.5 rounded-xl border ${theme.border} ${theme.card} hover:border-[#D97757] cursor-pointer transition flex items-center justify-between gap-3`}
                    >
                      <div className="min-w-0 flex-1">
                        <div className="flex items-center gap-2">
                          <span className="font-mono-claude text-xs font-bold text-[#D97757]">
                            Q{idx + 1}
                          </span>
                          {isFlagged && (
                            <span className="text-[10px] px-2 py-0.5 rounded-full bg-amber-500/20 text-amber-600 font-medium">
                              Review Flagged
                            </span>
                          )}
                          <span className={`text-xs truncate ${theme.textSecondary}`}>
                            {q.question}
                          </span>
                        </div>
                        {chosen !== undefined && (
                          <div className="text-xs font-medium mt-1 truncate">
                            Your Checked Answer: <span className="font-mono-claude font-semibold text-emerald-600 dark:text-emerald-400">Option ({optLetter})</span> â {q.options?.[chosen]}
                          </div>
                        )}
                      </div>

                      <div className="shrink-0 flex items-center gap-2">
                        {chosen !== undefined ? (
                          <span className="px-2.5 py-1 rounded-lg bg-emerald-500/15 text-emerald-600 dark:text-emerald-400 font-mono-claude text-xs font-semibold">
                            Option {optLetter}
                          </span>
                        ) : (
                          <span className={`px-2.5 py-1 rounded-lg border ${theme.border} text-xs ${theme.textSecondary}`}>
                            Skipped
                          </span>
                        )}
                        <ChevronRight className={`w-4 h-4 ${theme.textSecondary}`} />
                      </div>
                    </div>
                  );
                })}
              </div>

              {/* Modal Footer */}
              <div className={`px-5 py-4 border-t ${theme.border} ${theme.card} flex items-center justify-between gap-3`}>
                <button
                  onClick={() => setShowReviewDrawer(false)}
                  className={`px-4 py-2.5 rounded-xl border ${theme.border} ${theme.cardElevated} text-xs sm:text-sm font-medium`}
                >
                  Return to Questions
                </button>
                <button
                  onClick={handleSubmitQuizFinal}
                  className={`px-6 py-2.5 rounded-xl ${theme.accentBg} text-white text-xs sm:text-sm font-semibold shadow-md flex items-center gap-2`}
                >
                  <CheckCircle2 className="w-4 h-4" />
                  <span>Submit Final Paper ({attemptedCount}/{questions.length})</span>
                </button>
              </div>
            </div>
          </div>
        )}
      </div>
    );
  }

  // ==========================================
  // RENDER: MAIN MOBILE-FIRST APP LAYOUT
  // ==========================================
  return (
    <div className={`min-h-screen ${theme.bg} ${theme.textPrimary} flex flex-col font-sans-claude transition-colors duration-200`}>

           ========================================== */}

                          title: item.data.title,
                          subjectName: item.data.subjectName,
                          level: item.data.level,
                          negativeMarking: item.data.negativeMarking,
                          questions: item.data.questions
                        });
                        setQuizResult(item.data);
                      } else {
                        setViewingHistoryItem(item);
                      }
                    }}
                    className={`p-4 rounded-2xl border ${theme.border} ${theme.cardElevated} hover:border-[#D97757] cursor-pointer transition flex items-center justify-between gap-3`}

                    <div className="space-y-1 min-w-0">
                      <div className="flex items-center gap-2">
                        <span className={`text-[10px] font-mono-claude px-2 py-0.5 rounded border ${theme.accentSoft}`}>

                        </span>
                        <span className={`text-[11px] ${theme.textSecondary}`}>
                          {item.timestamp}
                        </span>
                      </div>
                      <div className="font-serif-claude text-base font-medium truncate">
                        {item.title}
                      </div>
                      <div className={`text-xs ${theme.textSecondary}`}>
                        {item.mode}
                      </div>
                    </div>

                    <div className="text-right shrink-0">
                      <div className="font-mono-claude text-base font-bold text-[#D97757]">

                      <div className={`text-[11px] font-mono-claude ${item.percentage >= 50 ? 'text-emerald-600 dark:text-emerald-400' : 'text-amber-600'}`}>

                      </div>

            TAB 4: SETTINGS & PHONE PWA INSTALLATION
           ========================================== */}

            className={`py-2 px-2 rounded-xl flex flex-col items-center justify-center gap-1 transition ${

  return (
    <div className={`p-5 sm:p-6 rounded-2xl border-2 border-[#D97757]/50 ${theme.cardElevated} shadow-lg space-y-6`}>

              ICAI Examiner Certified Evaluation
            </span>
          </div>
          <h3 className="font-serif-claude text-2xl font-medium mt-1">
            {report.paperTitle || 'Evaluated Answer Sheet Report'}
          </h3>
          <p className={`text-xs ${theme.textSecondary}`}>
            {report.examinerVerdict}
          </p>
        </div>

        <div className={`p-4 rounded-2xl border ${theme.border} ${theme.card} text-center min-w-[135px]`}>
          <div className="text-[11px] font-mono-claude uppercase text-[#D97757]">
            Marks Awarded
          </div>
          <div className="font-mono-claude text-3xl font-bold mt-0.5">
            {report.totalAwardedMarks}{' '}
            <span className={`text-base font-normal ${theme.textSecondary}`}>
              / {report.maxMarks}
            </span>
          </div>
          <div
            className={`text-xs font-semibold mt-0.5 ${
              report.percentage >= 40
                ? 'text-emerald-600 dark:text-emerald-400'
                : 'text-rose-500'
            }`}
          >
            {report.percentage}% ({report.percentage >= 60 ? 'Exemption Level!' : report.percentage >= 40 ? 'Pass (>=40%)' : 'Below 40% Criteria'})
          </div>
        </div>
      </div>

      {/* Strengths & ICAI Improvement Tips */}
      <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
        <div className={`p-3.5 rounded-xl border ${theme.border} ${theme.card} space-y-1.5`}>

          <div className="text-xs font-semibold text-[#D97757] flex items-center gap-1.5">
            <Award className="w-4 h-4" />
            <span>ICAI Examiner Improvement Advice</span>
          </div>
          <ul className={`text-xs ${theme.textSecondary} list-disc list-inside space-y-1`}>
            {(report.icaiImprovementTips || []).map((tip, i) => (
              <li key={i}>{tip}</li>
            ))}
          </ul>
        </div>
      </div>

      {/* Question-by-Question Step-Marking Breakdown */}

          Question-Wise Step Marking & ICAI Suggested Answers
        </h4>

            {qEval.specificDeductions && (

            <div className={`p-3 rounded-lg border ${theme.border} ${theme.cardElevated} text-xs space-y-1`}>

                <BookOpen className="w-3.5 h-3.5" />
                <span>Official ICAI Study Material / Suggested Answer:</span>
              </div>
              <p className="whitespace-pre-line leading-relaxed">{qEval.icaiSuggestedModelAnswer}</p>
            </div>
          </div>
        ))}
      </div>
    </div>
  );
